一、把所有英文遮住，只剩这次任务

第十六讲已经说明怎样验真。现在回到第一讲的请求：一个 Java 服务边界测试失败，先排查；获得修改授权后修复，再重新测试。你应能不依赖包名，说出信息如何进来、动作如何执行、事实如何留下、任务如何继续以及完成怎样证明。

这就是学习 DeepSeek Harness 的迁移目标。不同智能体常面对相近的输入、动作、状态与验证问题，但可能使用不同调度方式、记忆结构、工具协议和权限体系。共性在问题与责任，不在一定拥有同名类或使用相同插件框架。

二、最小运行器故意省掉什么

本书提供确定性的教学运行器：用内存文件替代真实磁盘，用预设调用替代真实模型，用金额比较的对应实验替代 Java 运行。它不接网络、不执行任意命令、不花模型费用。它能验证几条工程约束，不能评估模型推理质量。

Stub（测试替身）像排练时按固定台词说话的演员，帮助我们检查舞台机关和交接规则。真实模型可能给错参数、读错材料、反复尝试或过早结束；固定台词不会暴露所有这些问题。所以别把排练顺利叫作真实首演成功。

三、保留哪些最小责任

运行器保存调用与结果、一份可恢复快照、独立写许可、有限补丁规则和当前代码指纹。它拒绝未授权写入，拒绝旧版本补丁，不在恢复时重放外部动作，并要求新的测试结果才能验收。

这些职责与上游机制有对应关系，但没有实现 Cordis 插件生命周期、完整消息协议、持久化崩溃事务或真实沙箱。对照 主循环，你应能指出教学运行器省略了哪些部分，而非把它当另一份官方实现。

// Deterministic teaching model. No LLM, real shell, network or Java execution.
const { createHash } = require('node:crypto')
const digest = text => createHash('sha256').update(text).digest('hex')
const ORIGINAL = 'return orderCents > 10000 ? 0 : 800;'
const FIXED = 'return orderCents >= 10000 ? 0 : 800;'
const CONTRACT = '满 10000 分免运费；否则 800 分；不得改测试预期。'
const baseline = () => ({ code: ORIGINAL, contract: CONTRACT, report: 'amount=10000 expected=0 actual=800 FAIL' })

class TeachingHarness {
  constructor(files = baseline(), persisted = []) {
    this.files = files
    this.log = structuredClone(persisted)
    this.allowed = false // A new live executor requires fresh authorization.
    this.view = null
    // Close incomplete calls as unknown outcomes, never rerun them on recovery.
    const settled = new Set(this.log.filter(e => e.type === 'result').map(e => e.data.callId))
    for (const event of [...this.log]) if (event.type === 'call' && !settled.has(event.seq)) {
      this.append('result', { callId: event.seq, ok: false, error: 'interrupted; external outcome unknown' })
    }
  }
  append(type, data) {
    const event = { seq: this.log.length, type, data: structuredClone(data) }
    this.log.push(event)
    return event.seq
  }
  grant() {
    this.allowed = true
    this.append('authorization', { scope: 'one boundary fix; preserve tests' })
  }
  async execute(name, args = {}) {
    const callId = this.append('call', { name, args })
    let result
    try {
      if (name === 'read') {
        if (!Object.hasOwn(this.files, args.file)) throw Error('file not found')
        const text = this.files[args.file]
        result = { ok: true, file: args.file, text, hash: digest(text) }
      } else if (name === 'patch') {
        if (!this.allowed) throw Error('modification not authorized')
        if (args.expectedHash !== digest(this.files.code)) throw Error('code changed; read again')
        if (this.files.code !== ORIGINAL) throw Error('patch target no longer matches')
        this.files.code = FIXED
        this.allowed = false
        result = { ok: true, changed: true, hash: digest(this.files.code) }
      } else if (name === 'test') {
        if (![ORIGINAL, FIXED].includes(this.files.code)) throw Error('unknown teaching implementation')
        const rows = [9999, 10000, 10001].map(amount => {
          const actual = (this.files.code === FIXED ? amount >= 10000 : amount > 10000) ? 0 : 800
          return { amount, expected: amount >= 10000 ? 0 : 800, actual }
        })
        result = { ok: true, simulated: true, codeHash: digest(this.files.code), rows,
          exitCode: rows.some(r => r.actual !== r.expected) ? 1 : 0 }
      } else throw Error('unknown tool')
    } catch (error) { result = { ok: false, error: error.message } }
    this.append('result', { callId, ...result })
    return structuredClone(result)
  }
  compact(summary, sourceSeqs) {
    if (!Array.isArray(sourceSeqs) || !sourceSeqs.length || sourceSeqs.some(i => !Number.isInteger(i) || !this.log[i])) throw Error('missing source')
    const selected = sourceSeqs.map(i => this.log[i])
    for (const event of selected) {
      if (event.type === 'call' && !selected.some(e => e.type === 'result' && e.data.callId === event.seq)) throw Error('unpaired tool call')
      if (event.type === 'result' && !sourceSeqs.includes(event.data.callId)) throw Error('unpaired tool result')
    }
    this.append('summary', { text: summary, sourceSeqs })
    this.view = { text: summary, sourceSeqs: [...sourceSeqs] }
  }
  snapshot() { return structuredClone(this.log) }
  verify() {
    const results = this.log.filter(e => e.type === 'result').map(e => e.data)
    const has = file => results.some(r => r.ok && r.file === file)
    const latestTest = results.filter(r => r.simulated).at(-1)
    return { complete: has('report') && has('contract') && has('code')
      && this.log.some(e => e.type === 'authorization')
      && results.some(r => r.ok && r.changed)
      && this.files.code === FIXED && latestTest?.codeHash === digest(this.files.code)
      && latestTest.exitCode === 0 && latestTest.rows.length === 3,
    boundary: 'Only this deterministic fixture; not Java or a real agent evaluation.' }
  }
}
module.exports = { TeachingHarness, baseline, ORIGINAL, FIXED, digest }

const { TeachingHarness, digest } = require('./runner.cjs');
(async () => {
  const h = new TeachingHarness();
  console.log('unauthorized:', await h.execute('patch', {
    expectedHash: digest(h.files.code)
  }));
  for (const file of ['report', 'contract', 'code']) {
    await h.execute('read', { file });
  }
  console.log('before:', await h.execute('test'));
  h.grant();
  await h.execute('patch', { expectedHash: digest(h.files.code) });
  console.log('after:', await h.execute('test'));
  console.log('verification:', h.verify());
})();

先观察未授权请求返回错误，且源代码不变；原版三测在等于阈值时失败；显式授权后修改；新三测才通过。最后的验收对象会提醒你，这只是该确定性夹具，而不是 Java 或真实智能体评测。

五、自己制造四次“看起来成功”的假象

第一，把最后一次测试删掉，观察修改后的验收仍失败。第二，在测试通过后把代码改回原样，观察旧报告失效。第三，只保留读取报告，不读业务约定和代码，观察完整性验收失败。第四，将摘要写成“用户允许全部修改”，观察执行器仍拒绝未授权补丁。

再做一次恢复实验：向日志追加一个没有结果的编辑请求，创建新执行器恢复。它应记录未知结果而不修改文件。这个教学策略展示“恢复事实”和“重做动作”不同；不代表真实恢复后端的全部细节。

六、怎样把所学迁移到另一个框架

拿到另一个项目时，按六个问题找证据：谁构造模型输入；谁决定继续；工具请求在哪里真正执行并受约束；事实记录保存在哪里；中断后如何恢复；业务成功由什么证据证明。

再逐一记下不同选择。例如它未必采用仅追加日志，未必有独立子智能体，未必采用插件依赖激活。你要判断的是责任是否明确、缺口如何处理、约束能否通过反例验证，而不是强行将其类名翻译成这一套包名。

七、完整教材的终点与下一步实践

读完主课十七讲，应能从一条失败通知追到报告、实现、授权、补丁和新测试，并解释加入拒绝、恢复、压缩、分工后系统为何需要新机制。补课只帮助读语法和工程结构，不再打断主线编号。

下一次在真实测试项目里实践时，先选一个小任务、只读起步，明确权限和验收；再接真实模型与工具，补充端到端测试。把本书当作理解和实验的地图，不把它当作任何环境都安全可靠的保证。
